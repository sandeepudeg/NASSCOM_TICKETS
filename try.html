🔧 2. Low Level Design (LLD)

2.1 Core Service Components

Ticket Classification Pipeline

class TicketClassifier: def __init__(self, model_name: str, confidence_threshold: float = 0.65): self.ollama_client = OllamaClient() self.model_name = model_name self.confidence_threshold = confidence_threshold self.categories = [ "Infrastructure", "Application", "Security", "Database", "Storage", "Network", "Access Management" ] async def classify_ticket(self, ticket: TicketInput) -> ClassificationResult: # 1. PII Scrubbing scrubbed_content = await self.scrub_pii(ticket.content) # 2. Feature Extraction features = await self.extract_features(scrubbed_content) # 3. Model Inference prediction = await self.ollama_client.generate( model=self.model_name, prompt=self.build_classification_prompt(features), options={"temperature": 0.1, "top_p": 0.9} ) # 4. Confidence Scoring confidence = self.calculate_confidence(prediction) # 5. Escalation Logic if confidence < self.confidence_threshold: await self.escalate_ticket(ticket, confidence) return ClassificationResult( category=prediction.category, confidence=confidence, reasoning=prediction.reasoning, escalated=confidence < self.confidence_threshold )

RAG Resolution Engine

class RAGResolutionEngine: def __init__(self, vector_store: VectorStore, llm_client: LLMClient): self.vector_store = vector_store self.llm_client = llm_client self.similarity_threshold = 0.70 self.max_similar_tickets = 5 async def generate_resolution(self, ticket: Ticket) -> ResolutionResult: # 1. Vector Similarity Search similar_tickets = await self.vector_store.similarity_search( query_vector=ticket.embedding, threshold=self.similarity_threshold, limit=self.max_similar_tickets ) # 2. Context Preparation context = self.prepare_context(similar_tickets) # 3. Resolution Generation resolution_prompt = self.build_resolution_prompt(ticket, context) resolution = await self.llm_client.generate(resolution_prompt) # 4. Quality Scoring quality_score = await self.score_resolution_quality( ticket, resolution, similar_tickets ) return ResolutionResult( steps=resolution.steps, similar_tickets=similar_tickets, quality_score=quality_score, confidence=resolution.confidence )

2.2 Database Schema Design

Core Tables Structure

-- Tickets table with full-text search and vector embeddings CREATE TABLE tickets ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), title VARCHAR(255) NOT NULL, content TEXT NOT NULL, content_scrubbed TEXT NOT NULL, -- PII-scrubbed version category VARCHAR(50), priority VARCHAR(20) DEFAULT 'medium', status VARCHAR(20) DEFAULT 'open', confidence_score DECIMAL(3,2), embedding vector(768), -- pgvector for similarity search folder_id UUID REFERENCES folders(id), created_by UUID REFERENCES users(id), assigned_to UUID REFERENCES users(id), created_at TIMESTAMP DEFAULT NOW(), updated_at TIMESTAMP DEFAULT NOW(), resolved_at TIMESTAMP, -- Indexes for performance CONSTRAINT valid_confidence CHECK (confidence_score BETWEEN 0 AND 1), CONSTRAINT valid_status CHECK (status IN ('open', 'in_progress', 'resolved', 'closed', 'escalated')) ); -- Vector similarity index CREATE INDEX idx_tickets_embedding ON tickets USING ivfflat (embedding vector_cosine_ops); -- Full-text search index CREATE INDEX idx_tickets_fts ON tickets USING gin(to_tsvector('english', title || ' ' || content)); -- Classification results with model versioning CREATE TABLE classification_results ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), ticket_id UUID REFERENCES tickets(id) ON DELETE CASCADE, model_name VARCHAR(100) NOT NULL, model_version VARCHAR(50) NOT NULL, predicted_category VARCHAR(50) NOT NULL, confidence_score DECIMAL(3,2) NOT NULL, reasoning TEXT, processing_time_ms INTEGER, created_at TIMESTAMP DEFAULT NOW(), CONSTRAINT valid_confidence CHECK (confidence_score BETWEEN 0 AND 1) ); -- Resolution suggestions with RAG context CREATE TABLE resolution_suggestions ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), ticket_id UUID REFERENCES tickets(id) ON DELETE CASCADE, resolution_steps JSONB NOT NULL, similar_ticket_ids UUID[] NOT NULL, quality_score DECIMAL(3,2), llm_model VARCHAR(100) NOT NULL, context_used TEXT, created_at TIMESTAMP DEFAULT NOW(), CONSTRAINT valid_quality CHECK (quality_score BETWEEN 0 AND 1) ); -- Escalation tracking CREATE TABLE escalations ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), ticket_id UUID REFERENCES tickets(id) ON DELETE CASCADE, reason VARCHAR(100) NOT NULL, escalated_by VARCHAR(50) DEFAULT 'system', escalated_to UUID REFERENCES users(id), status VARCHAR(20) DEFAULT 'pending', notes TEXT, created_at TIMESTAMP DEFAULT NOW(), resolved_at TIMESTAMP, CONSTRAINT valid_escalation_status CHECK (status IN ('pending', 'in_review', 'resolved', 'dismissed')) );

📊 3. Data Sources & Engineering Pipeline

3.1 Primary Data Sources

🎯 Kaggle Dataset 1: IT Service Ticket Classification

Source: adisongoh/it-service-ticket-classification-dataset

Size: ~50,000 tickets across 7 categories

Format: CSV with columns: ticket_id, title, description, category, priority, status

Usage: Primary training data for classification model

# Data ingestion pipeline import kagglehub import pandas as pd def ingest_it_tickets(): path = kagglehub.dataset_download("adisongoh/it-service-ticket-classification-dataset") df = pd.read_csv(f"{path}/tickets.csv") # Data cleaning and validation df = df.dropna(subset=['title', 'description', 'category']) df['category'] = df['category'].str.title() # PII scrubbing df['description_scrubbed'] = df['description'].apply(scrub_pii) return df

🌍 Kaggle Dataset 2: Multilingual Customer Support Tickets

Source: tobiasbueck/multilingual-customer-support-tickets

Size: ~25,000 tickets in multiple languages

Format: JSON with nested structure for multilingual content

Usage: Enhance model robustness and language coverage

🚨 Kaggle Dataset 3: ServiceNow-like Incident Response Log

Source: vipulshinde/incident-response-log

Size: ~15,000 incident records with resolution steps

Format: CSV with incident details and resolution workflows

Usage: Training data for RAG resolution generation

3.2 Real-time Data Ingestion

📥 Structured Log Ingestion

Sources: ECS JSON logs, OpenTelemetry traces, Prometheus alerts

Processing: Real-time stream processing with Apache Kafka

# Kafka consumer for real-time ticket ingestion from kafka import KafkaConsumer import json class TicketIngestionConsumer: def __init__(self): self.consumer = KafkaConsumer( 'ticket-events', bootstrap_servers=['kafka:9092'], value_deserializer=lambda x: json.loads(x.decode('utf-8')) ) async def process_messages(self): for message in self.consumer: ticket_data = message.value # Extract causal context from telemetry context = self.extract_causal_context(ticket_data) # Create ticket with enhanced metadata ticket = await self.create_ticket_with_context(ticket_data, context) # Trigger async classification await self.classify_ticket_async(ticket)

# Complete data engineering pipeline class DataEngineeringPipeline: def __init__(self): self.presidio_analyzer = PresidioAnalyzer() self.embedding_model = SentenceTransformer('all-MiniLM-L6-v2') self.vector_store = QdrantClient() async def process_dataset(self, dataset_path: str) -> ProcessingResult: # 1. Data Validation df = self.validate_and_clean_data(dataset_path) # 2. PII Scrubbing df['content_scrubbed'] = df['content'].apply(self.scrub_pii) # 3.
